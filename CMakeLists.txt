cmake_minimum_required(VERSION 3.16)
project(ExactSearch VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

function(exactsearch_options target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  endif()
endfunction()

add_library(exactsearch_core src/search.cpp src/io.cpp src/batch.cpp)
target_include_directories(exactsearch_core PUBLIC include)
target_link_libraries(exactsearch_core PUBLIC Threads::Threads)
exactsearch_options(exactsearch_core)
foreach(target exactsearch exactsearch_profile exactsearch_batch)
  if(target STREQUAL "exactsearch")
    set(source src/main.cpp)
  elseif(target STREQUAL "exactsearch_profile")
    set(source src/profile.cpp)
  else()
    set(source src/batch_bench.cpp)
  endif()
  add_executable(${target} ${source} src/entry.cpp)
  target_link_libraries(${target} PRIVATE exactsearch_core)
  exactsearch_options(${target})
  if(MINGW)
    target_link_options(${target} PRIVATE -municode)
  endif()
endforeach()

include(CTest)
if(BUILD_TESTING)
  add_executable(exactsearch_tests tests/test_search.cpp tests/test_io_batch.cpp tests/test_batch.cpp)
  target_link_libraries(exactsearch_tests PRIVATE exactsearch_core)
  exactsearch_options(exactsearch_tests)
  # Existing I/O fixtures use build/ relative to the source root.
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/build")
  add_test(NAME core_correctness COMMAND exactsearch_tests)
  set_tests_properties(core_correctness PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME cli_regression COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/check_cli.py")
    set_tests_properties(cli_regression PROPERTIES ENVIRONMENT "EXACTSEARCH_BUILD_DIR=${CMAKE_CURRENT_BINARY_DIR}")
    add_test(NAME benchmark_snapshot COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/audit_snapshot.py")
  endif()
endif()
